        frameCount = 0

    end

local CAM_PREDICTION = 0.12

local function CreateRedHighlight()
    if redHighlight then redHighlight:Destroy() end
    redHighlight = Instance.new("Highlight")
    redHighlight.FillColor = Color3.fromRGB(255, 0, 0)
    redHighlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    redHighlight.FillTransparency = 0.1
    redHighlight.OutlineTransparency = 0.2
    redHighlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    redHighlight.Enabled = false
    redHighlight.Parent = game:GetService("CoreGui")
end

local function CreateLineDrawing()
    if lineDrawing then lineDrawing:Remove() end
    lineDrawing = Drawing.new("Line")
    lineDrawing.Color = Color3.fromRGB(255, 0, 0)
    lineDrawing.Thickness = 2
    lineDrawing.Transparency = 0.8
    lineDrawing.Visible = false
end

local function RemoveVisuals()
    if redHighlight then
        redHighlight.Enabled = false
        redHighlight.Adornee = nil
    end
    if lineDrawing then
        lineDrawing.Visible = false
    end
end

local function UpdateVisuals(targetPlayer)
    if not redHighlight then CreateRedHighlight() end
    if targetPlayer and targetPlayer.Character then
        redHighlight.Adornee = targetPlayer.Character
        redHighlight.Enabled = true
    else
        redHighlight.Enabled = false
        redHighlight.Adornee = nil
    end
end

local function UpdateLineDrawing(aimPos)
    if not lineDrawing then CreateLineDrawing() end
    if not aimPos or not (SilentAimEnabled or CamLockEnabled) then
        lineDrawing.Visible = false
        return
    end
    local cam = Camera
    if not cam then
        lineDrawing.Visible = false
        return
    end
    local targetScreenPos, targetOnScreen = cam:WorldToScreenPoint(aimPos)
    if targetOnScreen then
        local screenSize = cam.ViewportSize
        local startPos = Vector2.new(screenSize.X / 2, 0)
        lineDrawing.From = startPos
        lineDrawing.To = Vector2.new(targetScreenPos.X, targetScreenPos.Y)
        lineDrawing.Visible = true
    else
        lineDrawing.Visible = false
    end
end

local function Notify(title, content, icon, duration)
    if type(WindUI.Notify) == "function" then
        WindUI:Notify({ Title = title, Content = content, Icon = icon or "info", Duration = duration or 2 })
    end
end

local function GetPlayerNames()
    local names = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then table.insert(names, p.Name) end
    end
    table.sort(names)
    return names
end

local function GetPlayerByName(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name == name and p ~= LocalPlayer then return p end
    end
    return nil
end

local function GetClosestPlayerByDistance()
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local root = char.HumanoidRootPart
    local nearest, minDist = nil, LOCK_DISTANCE
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
        local hum = p.Character and p.Character:FindFirstChildOfClass("Humanoid")
        if hrp and hum and hum.Health > 0 then
            local d = (hrp.Position - root.Position).Magnitude
            if d < minDist then
                minDist = d
                nearest = p
            end
        end
    end
    return nearest
end

local function CamIsValid(plr)
    if not plr or not plr.Character then return false end
    local hum = plr.Character:FindFirstChildOfClass("Humanoid")
    local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return false end
    if hum.Health <= 0 then return false end
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return false end
    return (hrp.Position - myChar.HumanoidRootPart.Position).Magnitude <= CAM_LOCK_DISTANCE
end

local function CamFindTarget()
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local root = myChar.HumanoidRootPart
    local nearest, bestDist = nil, CAM_LOCK_DISTANCE
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local hum = plr.Character:FindFirstChildOfClass("Humanoid")
            local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
            if hum and hrp and hum.Health > 0 then
                local d = (hrp.Position - root.Position).Magnitude
                if d < bestDist then
                    bestDist = d
                    nearest = plr
                end
            end
        end
    end
    return nearest
end

local aimCamConnection = nil

local function aimCamUpdate()
    if SilentAimEnabled then
        local char = LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            if not CurrentTarget
                or not CurrentTarget.Character
                or not CurrentTarget.Character:FindFirstChild("Humanoid")
                or CurrentTarget.Character.Humanoid.Health <= 0
            then
                CurrentTarget = GetClosestPlayerByDistance()
            end

            if CurrentTarget and CurrentTarget.Character then
                local targetHRP = CurrentTarget.Character:FindFirstChild("HumanoidRootPart")
                if targetHRP then
                    char.HumanoidRootPart.CFrame = CFrame.new(
                        char.HumanoidRootPart.Position,
                        Vector3.new(targetHRP.Position.X, char.HumanoidRootPart.Position.Y, targetHRP.Position.Z)
                    )
                    UpdateVisuals(CurrentTarget)
                    local aimPos = targetHRP.Position + targetHRP.AssemblyLinearVelocity * CAM_PREDICTION
                    UpdateLineDrawing(aimPos)
                end
            else
                RemoveVisuals()
                CurrentTarget = nil
            end
        end
    else
        CurrentTarget = nil
    end

    if CamLockEnabled then
        if not CamIsValid(CamlockTarget) then
            if tick() - CamLastValid < CamGrace then return end
            CamlockTarget = CamFindTarget()
            if not CamlockTarget then
                if not SilentAimEnabled then RemoveVisuals() end
                return
            end
        end

        CamLastValid = tick()
        local hrp = CamlockTarget.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            local predicted = hrp.Position + hrp.AssemblyLinearVelocity * CAM_PREDICTION
            local currentCF = Camera.CFrame
            local targetCF = CFrame.new(currentCF.Position, predicted)
            Camera.CFrame = currentCF:Lerp(targetCF, CAM_SMOOTHNESS)

            UpdateVisuals(CamlockTarget)
            UpdateLineDrawing(predicted)
        end
    else
        CamlockTarget = nil
    end

    if not SilentAimEnabled and not CamLockEnabled then
        RemoveVisuals()
    end
end

local function safeSetSilentToggle(value)
    if not toggleSilent then return end
    suppressSilentCallback = true
    pcall(function() toggleSilent:SetValue(value) end)
    suppressSilentCallback = false
end

local function safeSetCamToggle(value)
    if not toggleCam then return end
    suppressCamCallback = true
    pcall(function() toggleCam:SetValue(value) end)
    suppressCamCallback = false
end

local function updateAimCamConnection()
    if SilentAimEnabled or CamLockEnabled then
        if not aimCamConnection then
            aimCamConnection = RunService.RenderStepped:Connect(aimCamUpdate)
        end
    else
        if aimCamConnection then
            aimCamConnection:Disconnect()
            aimCamConnection = nil
        end
        RemoveVisuals()
    end
end

local function SetSilentAim(value)
    SilentAimEnabled = value
    updateAimCamConnection()
end

local function SetCamLock(value)
    CamLockEnabled = value
    updateAimCamConnection()
end

local function GetPlayerFromTarget(target)
    if not target then return nil end
    local model = target:FindFirstAncestorOfClass("Model")

    if not model:FindFirstChildOfClass("Humanoid") then return nil end

function OrbitController:Start(plr)
    if not plr or plr == LocalPlayer then return false end
    if self.Enabled then self:Stop() end

    local localHRP = GetHRP(LocalPlayer)
    if not localHRP then
        local character = LocalPlayer.Character
        if character then
            character:WaitForChild("Humanoid", 5)
            localHRP = character:WaitForChild("HumanoidRootPart", 5)
        end
    end

    local targetHRP = GetHRP(plr)
    if not localHRP or not targetHRP then
        warn("[Orbit] Không tìm thấy HRP")
        return false
    end

    self.Enabled = true
    self.Target = plr

    local offsetVec = localHRP.Position - targetHRP.Position
    if offsetVec.Magnitude > 0.01 then
        self.Angle = math.atan2(offsetVec.Z, offsetVec.X)
    else
        self.Angle = 0
    end

    if self.CameraPart then self.CameraPart:Destroy() end
    self.CameraPart = Instance.new("Part")
    self.CameraPart.Name = "OrbitCameraFocus"
    self.CameraPart.Size = Vector3.new(0.1, 0.1, 0.1)
    self.CameraPart.Transparency = 1
    self.CameraPart.CanCollide = false
    self.CameraPart.Anchored = true
    self.CameraPart.CFrame = CFrame.new(targetHRP.Position)
    self.CameraPart.Parent = Workspace

    local cam = Workspace.CurrentCamera
    if cam then
        cam.CameraType = Enum.CameraType.Custom
        cam.CameraSubject = self.CameraPart
    end

    local targetPos = targetHRP.Position
    local offset = Vector3.new(math.cos(self.Angle) * Radius, Height, math.sin(self.Angle) * Radius)
    local startPos = targetPos + offset
    localHRP.CFrame = CFrame.lookAt(startPos, targetPos)
    localHRP.AssemblyLinearVelocity = Vector3.zero
    localHRP.AssemblyAngularVelocity = Vector3.zero

    self:BeginOrbitLoop(plr)
    return true
end

function OrbitController:BeginOrbitLoop(plr)
    if not self.Enabled then return end

    if self.CameraConnection then self.CameraConnection:Disconnect(); self.CameraConnection = nil end
    if self.PhysicsConnection then self.PhysicsConnection:Disconnect(); self.PhysicsConnection = nil end
    if self.LineConnection then self.LineConnection:Disconnect(); self.LineConnection = nil end

    self.CameraConnection = RunService.RenderStepped:Connect(function(dt)
        if not self.Enabled then return end
        local target = self.Target
        if not target then return end
        local targetRoot = GetHRP(target)
        if not targetRoot then return end

        self.Angle = (self.Angle + AngleSpeed * dt) % (math.pi * 2)

        local targetPosition = targetRoot.Position
        local offset = Vector3.new(math.cos(self.Angle) * Radius, Height, math.sin(self.Angle) * Radius)
        local desiredPosition = targetPosition + offset

        OrbitState.TargetPosition = desiredPosition
        OrbitState.TargetLookAt = targetPosition

        if self.CameraPart then
            self.CameraPart.CFrame = CFrame.new(targetPosition)
        end
    end)

    self.PhysicsConnection = RunService.Heartbeat:Connect(function()
        if not self.Enabled then return end
        local myRoot = GetHRP(LocalPlayer)
        local targetRoot = GetHRP(self.Target)
        if not myRoot or not targetRoot then return end

        local desiredPosition = targetRoot.Position + Vector3.new(
            math.cos(self.Angle) * Radius,
            Height,
            math.sin(self.Angle) * Radius
        )
        local lookAt = targetRoot.Position

        myRoot.CFrame = CFrame.lookAt(desiredPosition, lookAt)

        myRoot.AssemblyLinearVelocity = Vector3.zero
        myRoot.AssemblyAngularVelocity = Vector3.zero

        if RemoteEvent then
            RemoteEvent:FireServer(nil, myRoot.CFrame)
        end
    end)

    if not self.Line then
        self.Line = Drawing.new("Line")
        self.Line.Thickness = 2
        self.Line.Color = Color3.fromRGB(255, 0, 0)
        self.Line.Transparency = 1
        self.Line.Visible = false
    end

    self.LineConnection = RunService.RenderStepped:Connect(function()
        if not self.Enabled or not self.Target then
            self.Line.Visible = false
            return
        end
        local targetRoot = GetHRP(self.Target)
        local myRoot = GetHRP(LocalPlayer)
        local cam = Workspace.CurrentCamera
        if not targetRoot or not myRoot or not cam then
            self.Line.Visible = false
            return
        end
        local myPos, on1 = cam:WorldToViewportPoint(myRoot.Position)
        local tgtPos, on2 = cam:WorldToViewportPoint(targetRoot.Position)
        if on1 and on2 then
            self.Line.Visible = true
            self.Line.From = Vector2.new(myPos.X, myPos.Y)
            self.Line.To = Vector2.new(tgtPos.X, tgtPos.Y)
        else
            self.Line.Visible = false
        end
    end)
end

function OrbitController:Stop()
    self.Enabled = false
    self.Target = nil

    if self.PhysicsConnection then
        self.PhysicsConnection:Disconnect()
        self.PhysicsConnection = nil
    end
    OrbitState.TargetPosition = nil
    OrbitState.TargetLookAt = nil

    if self.CameraConnection then self.CameraConnection:Disconnect(); self.CameraConnection = nil end
    if self.LineConnection then self.LineConnection:Disconnect(); self.LineConnection = nil end
    if self.Line then self.Line.Visible = false end
    if self.CameraPart then self.CameraPart:Destroy(); self.CameraPart = nil end

    local cam = Workspace.CurrentCamera
    if cam then
        local character = LocalPlayer.Character
        if character then
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            cam.CameraType = Enum.CameraType.Custom
            if humanoid then cam.CameraSubject = humanoid end
        else
            cam.CameraSubject = nil
        end
    end
    LocalPlayer.DevCameraOcclusionMode = defaultOcclusion
end

function OrbitController:DisableMouseSelection()
    if self.MouseMoveConnection then self.MouseMoveConnection:Disconnect(); self.MouseMoveConnection = nil end
    if self.MouseClickConnection then self.MouseClickConnection:Disconnect(); self.MouseClickConnection = nil end
    if self.MouseHighlight then self.MouseHighlight:Destroy(); self.MouseHighlight = nil end
    Mouse.Icon = ""
end

    Running = false,
    Cycle = 0,

local function GetCharacter()
    local character = Player.Character

    character = character or GetCharacter()

    character = character or GetCharacter()

local function DistanceToTarget(root)
    if not root then
        return math.huge
    end
    return (root.Position - CONFIG.TargetPosition).Magnitude
end

    return DistanceToTarget(root) <= CONFIG.PositionTolerance
end

    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = title,
            Text = text,
            Duration = 2
        })
    end)

local function UpdateCharacter(character)
    State.Character = character
    State.Humanoid = nil
    State.RootPart = nil
    if not character then
        return
    end
    State.Humanoid = character:FindFirstChildOfClass("Humanoid")
    State.RootPart = character:FindFirstChild("HumanoidRootPart")
end

local function FireReset()
    local success = pcall(function()
        local Knit = ReplicatedStorage:WaitForChild("Knit", 5)
        local KnitCore = Knit:WaitForChild("Knit", 5)
        local Services = KnitCore:WaitForChild("Services", 5)
        local JoinService = Services:WaitForChild("JoinService", 5)
        local RE = JoinService:WaitForChild("RE", 5)
        local Reset = RE:WaitForChild("Reset", 5)
        if not Reset:IsA("RemoteEvent") then
            error("Reset is not RemoteEvent")
        end
        Reset:FireServer()
    end)
    return success
end

local function WaitForNewCharacter(oldCharacter)
    local newCharacter
    if Player.Character and Player.Character ~= oldCharacter then
        newCharacter = Player.Character
    else
        local connection
        local finished = false
        connection = Player.CharacterAdded:Connect(function(character)
            if not finished then
                finished = true
                newCharacter = character
            end
        end)
        local start = os.clock()
        while not newCharacter and os.clock() - start < 10 do
            if not State.Enabled then
                break
            end
            task.wait(0.05)
        end
        finished = true
        connection:Disconnect()
    end
    if not newCharacter then
        return nil
    end
    local humanoid = newCharacter:WaitForChild("Humanoid", 10)
    local root = newCharacter:WaitForChild("HumanoidRootPart", 10)
    if not humanoid or not root then
        return nil
    end
    local start = os.clock()
    while humanoid.Health <= 0 and os.clock() - start < 5 do
        if not State.Enabled then
            return nil
        end
        task.wait(0.05)
    end
    if humanoid.Health <= 0 then
        return nil
    end
    UpdateCharacter(newCharacter)
    return newCharacter
end

local function TweenToTarget(character)
    local humanoid = GetHumanoid(character)
    local root = GetRoot(character)
    if not humanoid or not root then
        return false
    end
    if humanoid.Health <= 0 then
        return false
    end
    local target = CFrame.new(CONFIG.TargetPosition)
    local tweenInfo = TweenInfo.new(CONFIG.TweenTime, Enum.EasingStyle.Linear, Enum.EasingDirection.Out, 0, false, 0)
    local success = pcall(function()
        local tween = TweenService:Create(root, tweenInfo, { CFrame = target })
        tween:Play()
        tween.Completed:Wait()
    end)
    return success
end

local function ValidateStable(character)
    local humanoid = GetHumanoid(character)
    local root = GetRoot(character)
    if not humanoid or not root then
        return false
    end
    local startTime = os.clock()
    while os.clock() - startTime < CONFIG.StableTime do
        if not State.Enabled then
            return false
        end
        if Player.Character ~= character then
            return false
        end
        if humanoid.Health <= 0 then
            return false
        end
        if not IsAtTarget(root) then
            return false
        end
        task.wait(CONFIG.CheckInterval)
    end
    if Player.Character ~= character then
        return false
    end
    if humanoid.Health <= 0 then
        return false
    end
    return IsAtTarget(root)
end

local function ExecuteCycle()
    State.Cycle += 1
    local oldCharacter = GetCharacter()
    local oldHumanoid = GetHumanoid(oldCharacter)
    if not oldCharacter or not oldHumanoid or oldHumanoid.Health <= 0 then
        return false
    end
    if not FireReset() then
        return false
    end
    local newCharacter = WaitForNewCharacter(oldCharacter)
    if not newCharacter then
        return false
    end
    local delayStart = os.clock()
    while os.clock() - delayStart < CONFIG.RespawnDelay do
        if not State.Enabled then
            return false
        end
        local humanoid = GetHumanoid(newCharacter)
        if not humanoid or humanoid.Health <= 0 then
            return false
        end
        task.wait(0.05)
    end
    if not TweenToTarget(newCharacter) then
        return false
    end
    local root = GetRoot(newCharacter)
    if not root then
        return false
    end
    if not IsAtTarget(root) then
        return false
    end
    return ValidateStable(newCharacter)
end

local function StartController()
    if State.Running then
        return
    end
    State.Running = true
    State.Cycle = 0
    task.spawn(function()
        while State.Enabled do
            local success = ExecuteCycle()
            if success then
                State.Enabled = false
                Notify("✓ Auto Safe Reset", "Vị trí ổn định trong 3 giây.")
                break
            end
            if State.Enabled then
                task.wait(CONFIG.RetryDelay)
            end
        end
        State.Running = false
    end)
end

            end)
        )
    end

    HoldButton(upButton, function(state)
        CameraSystem.UpPressed = state
    end)
    HoldButton(downButton, function(state)
        CameraSystem.DownPressed = state
    end)

    local activeLookTouch = nil
    local lastTouchPosition = nil

    local function IsTouchOnControl(pos)
        return IsPointInGui(pos, joystickBase)
            or IsPointInGui(pos, upButton)
            or IsPointInGui(pos, downButton)
    end

    table.insert(CameraSystem.TouchConnections,
        lookArea.InputBegan:Connect(function(input)
            if input.UserInputType ~= Enum.UserInputType.Touch then
                return
            end
            local pos = Vector2.new(input.Position.X, input.Position.Y)
            if IsTouchOnControl(pos) then
                return
            end
            activeLookTouch = input
            lastTouchPosition = input.Position
        end)
    )

    table.insert(CameraSystem.TouchConnections,
        lookArea.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch and input == activeLookTouch then
                if lastTouchPosition then
                    local delta = input.Position - lastTouchPosition
                    CameraSystem.Yaw -= delta.X * TOUCH_SENSITIVITY
                    CameraSystem.Pitch -= delta.Y * TOUCH_SENSITIVITY
                    ClampPitch()
                end
                lastTouchPosition = input.Position
            end
        end)
    )

    table.insert(CameraSystem.TouchConnections,
        UserInputService.TouchEnded:Connect(function(input)
            if input == activeLookTouch then
                activeLookTouch = nil
                lastTouchPosition = nil
            end
        end)
    )
end

local function EnableFreeCamera()
    if CameraSystem.Enabled then return end
    local Camera = GetCamera()
    local part = CreateTargetPart()
    if not Camera or not part then return end
    CameraSystem.Enabled = true
    CameraSystem.SavedCameraType = Camera.CameraType
    CameraSystem.SavedCameraSubject = Camera.CameraSubject
    CameraSystem.SavedCameraCFrame = Camera.CFrame
    CameraSystem.Position = part.Position
    CameraSystem.Yaw = START_YAW
    CameraSystem.Pitch = START_PITCH
    table.clear(CameraSystem.Keys)
    CameraSystem.UpPressed = false
    CameraSystem.DownPressed = false
    CameraSystem.MoveInput = Vector2.zero
    Camera.CameraType = Enum.CameraType.Scriptable

    CameraSystem.InputBegan = UserInputService.InputBegan:Connect(function(input, processed)
        if not CameraSystem.Enabled or processed then return end
        if input.UserInputType == Enum.UserInputType.Keyboard then
            CameraSystem.Keys[input.KeyCode] = true
        end
    end)

    CameraSystem.InputEnded = UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Keyboard then
            CameraSystem.Keys[input.KeyCode] = false
        end
    end)

    CameraSystem.InputChanged = UserInputService.InputChanged:Connect(function(input)
        if not CameraSystem.Enabled then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement then
            CameraSystem.Yaw -= input.Delta.X * LOOK_SENSITIVITY
            CameraSystem.Pitch -= input.Delta.Y * LOOK_SENSITIVITY
            ClampPitch()
        end
    end)

    CreateMobileControls()

    if not CameraSystem.RenderBound then
        RunService:BindToRenderStep(
            "KhanhlyFreeCamera",
            Enum.RenderPriority.Camera.Value + 1,
            function(deltaTime)
                if not CameraSystem.Enabled then return end
                local currentCamera = GetCamera()
                if not currentCamera then return end
                local rotation = GetCameraRotation()
                local movement = GetMovementDirection()
                if movement.Magnitude > 0 then
                    local worldDirection = rotation:VectorToWorldSpace(movement)
                    local speed = CameraSystem.Keys[Enum.KeyCode.LeftShift] and SPRINT_SPEED or MOVE_SPEED
                    CameraSystem.Position += worldDirection * speed * deltaTime
                end
                currentCamera.CameraType = Enum.CameraType.Scriptable
                currentCamera.CFrame = CFrame.new(CameraSystem.Position) * rotation
            end
        )
        CameraSystem.RenderBound = true
    end
end

local function DisableFreeCamera()
    if not CameraSystem.Enabled then return end
    CameraSystem.Enabled = false
    if CameraSystem.RenderBound then
        pcall(function()
            RunService:UnbindFromRenderStep("KhanhlyFreeCamera")
        end)
        CameraSystem.RenderBound = false
    end
    ClearConnections()
    table.clear(CameraSystem.Keys)
    CameraSystem.MoveInput = Vector2.zero
    CameraSystem.UpPressed = false
    CameraSystem.DownPressed = false
    if CameraSystem.MobileGui then
        CameraSystem.MobileGui:Destroy()
        CameraSystem.MobileGui = nil
    end
    local Camera = GetCamera()
    if Camera then
        local humanoid = GetCharacterHumanoid()
        Camera.CameraType = Enum.CameraType.Custom
        if humanoid then
            Camera.CameraSubject = humanoid
        end
    end
    CameraSystem.Position = nil
end

local function isDummyCharacter(character)
    if not character then return false end
    local container = Workspace:FindFirstChild("Characters")
    if not container then return false end
    local dummy = container:FindFirstChild("Dummy")
    if not dummy then return false end
    return character == dummy or character:IsDescendantOf(dummy)

local function getTargets()
    local targets = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            if not isDummyCharacter(plr.Character) then
                local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
                local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    table.insert(targets, hrp)
                end
            end
        end
    end
    return targets
end

local function findNearestTarget(fromPos)
    local nearest, minDist = nil, math.huge
    for _, target in ipairs(getTargets()) do
        if target and target.Parent then
            local dist = (target.Position - fromPos).Magnitude
            if dist < minDist then
                nearest, minDist = target, dist
            end
        end
    end
    return nearest, minDist
end

local function chooseTarget(mode, nameFilter)
    local targets = getTargets()
    if #targets == 0 then return nil end

    if mode == "Random" then
        return targets[math.random(1, #targets)]
    end

    if mode == "Nearest" then
        local myHRP = getHRP()
        if not myHRP then return nil end
        return findNearestTarget(myHRP.Position)
    end

    if mode == "Specified" and nameFilter and nameFilter ~= "" then
        local needle = nameFilter:lower()
        for _, target in ipairs(targets) do
            local plr  = Players:GetPlayerFromCharacter(target.Parent)
            local name = plr and plr.Name or target.Parent.Name
            if name:lower():find(needle, 1, true) then
                return target
            end
        end
    end
    return nil
end

-- Main update loop
local function UpdateESP()
    if not ESPEnabled then return end

    local folder = RefreshCharactersFolder()
    if not folder then return end

    local localChar = LocalPlayer.Character
    if not localChar then return end
    local localRoot = localChar:FindFirstChild("HumanoidRootPart")
    if not localRoot then return end

    local screenSize = Camera.ViewportSize
    local snapOrigin = Vector2.new(screenSize.X / 2, screenSize.Y - 2)

    local seenPlayers = {}

    for _, player in ipairs(Players:GetPlayers()) do
        seenPlayers[player] = true

        -- Bỏ qua chính mình
        if player == LocalPlayer then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        -- Lấy character thật
        local char = GetRealCharacter(player)
        if not char then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local rootPart = char:FindFirstChild("HumanoidRootPart")
        local head = char:FindFirstChild("Head")
        if not humanoid or not rootPart or not head then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        if humanoid.Health <= 0 then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        local distance = (rootPart.Position - localRoot.Position).Magnitude
        if distance > MAX_DISTANCE then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        -- Tạo/lấy data
        local data = EspData[player]
        if not data then
            data = CreatePlayerESP(player)
        end

        -- Hiện container
        data.container.Visible = true

        -- Highlight (chỉ đổi khi character thay đổi)
        if data.character ~= char then
            data.character = char
            data.highlight.Adornee = char
            data.highlight.Enabled = true
        elseif not data.highlight.Enabled then
            data.highlight.Enabled = true
        end

        -- SnapLine (Drawing.Line)
        local rootScreen = Camera:WorldToViewportPoint(rootPart.Position)
        if rootScreen.Z > 0 then
            local targetX = math.clamp(rootScreen.X, 0, screenSize.X)
            local targetY = math.clamp(rootScreen.Y, 0, screenSize.Y)
            data.snapLine.Visible = true
            data.snapLine.From = snapOrigin
            data.snapLine.To = Vector2.new(targetX, targetY)
        else
            data.snapLine.Visible = false
        end

        -- Tính toán Box/Health/Name/Distance
        local headPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
        local hipHeight = humanoid.HipHeight or 2
        local feetPos = Camera:WorldToViewportPoint(rootPart.Position - Vector3.new(0, hipHeight + 2, 0))

        if headPos.Z <= 0 and feetPos.Z <= 0 then
            data.box.Visible = false
            data.outline.Visible = false
            data.bgBar.Visible = false
            data.healthBar.Visible = false
            data.nameText.Visible = false
            data.distText.Visible = false
            continue
        end

        local top = math.min(headPos.Y, feetPos.Y)
        local bottom = math.max(headPos.Y, feetPos.Y)
        local height = math.max(bottom - top, 20)
        local width = math.max(height * 0.6, 24)
        local left = headPos.X - width / 2

        top = math.clamp(top, -50, screenSize.Y + 50)
        left = math.clamp(left, -50, screenSize.X + 50)

        -- Box
        data.box.Visible = true
        data.box.Position = UDim2.fromOffset(left, top)
        data.box.Size = UDim2.fromOffset(width, height)

        -- Outline
        data.outline.Visible = true
        data.outline.Position = UDim2.fromOffset(left - 2, top - 2)
        data.outline.Size = UDim2.fromOffset(width + 4, height + 4)

        -- Health bar
        local maxHealth = math.max(humanoid.MaxHealth, 1)
        local healthPercent = math.clamp(humanoid.Health / maxHealth, 0, 1)
        local barWidth = math.clamp(height / 25, 2, 4)
        local barLeft = left - barWidth - 6

        data.bgBar.Visible = true
        data.bgBar.Position = UDim2.fromOffset(barLeft, top)
        data.bgBar.Size = UDim2.fromOffset(barWidth, height)

        local fillHeight = height * healthPercent
        data.healthBar.Visible = true
        data.healthBar.Position = UDim2.fromOffset(barLeft, top + height - fillHeight)
        data.healthBar.Size = UDim2.fromOffset(barWidth, fillHeight)
        data.healthBar.BackgroundColor3 = Color3.fromRGB(
            255 * (1 - healthPercent),
            255 * healthPercent,
            0
        )

        -- Name
        data.nameText.Visible = true
        data.nameText.Text = player.Name
        data.nameText.Position = UDim2.fromOffset(headPos.X, top - 2)

        -- Distance
        data.distText.Visible = true
        data.distText.Text = string.format("M:%03d", math.floor(distance))
        data.distText.Position = UDim2.fromOffset(headPos.X, top + 16)
    end

    -- Xoá player đã rời
    for player in pairs(EspData) do
        if not seenPlayers[player] then
            ClearPlayerESP(player)
        end
    end
end

SttTab:Section({
    Title = "Fly"

local player = game.Players.LocalPlayer
local camera = workspace.CurrentCamera
local runService = game:GetService("RunService")
local uis = game:GetService("UserInputService")

local flying = false
local flySpeed = 50
local flyConnection = nil
local controls = nil

pcall(function()
    local playerModule = require(player:WaitForChild("PlayerScripts"):WaitForChild("PlayerModule"))
    controls = playerModule:GetControls()
end)

local function getMoveVector()
    if controls then
        return controls:GetMoveVector()
    end
    local move = Vector3.zero
    if uis:IsKeyDown(Enum.KeyCode.W) then move += Vector3.new(0, 0, -1) end
    if uis:IsKeyDown(Enum.KeyCode.S) then move += Vector3.new(0, 0, 1) end
    if uis:IsKeyDown(Enum.KeyCode.A) then move += Vector3.new(-1, 0, 0) end
    if uis:IsKeyDown(Enum.KeyCode.D) then move += Vector3.new(1, 0, 0) end
    if uis:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
    if uis:IsKeyDown(Enum.KeyCode.LeftShift) then move += Vector3.new(0, -1, 0) end
    return move
end

local function startFly()
    if flying then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    flying = true

    hum.AutoRotate = false
    hum.PlatformStand = true

    local bg = Instance.new("BodyGyro")
    bg.P = 9e4
    bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bg.Parent = root

    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.Velocity = Vector3.zero
    bv.Parent = root

    local defaultFOV = camera.FieldOfView
    local maxFOV = 105
    local fovSmooth = 0.1

    flyConnection = runService.Heartbeat:Connect(function()
        if not flying or not char.Parent or not root.Parent or not hum.Parent then
            stopFly()
            return
        end

        local input = getMoveVector()
        local desiredVelocity

        if controls then
            local camRelative = Vector3.new(input.X, input.Y, input.Z)
            desiredVelocity = camera.CFrame:VectorToWorldSpace(camRelative) * flySpeed
        else
            local camLook = camera.CFrame.LookVector
            local camRight = camera.CFrame.RightVector
            local forward = Vector3.new(camLook.X, 0, camLook.Z).Unit
            local right = Vector3.new(camRight.X, 0, camRight.Z).Unit

            desiredVelocity = (right * input.X + forward * (-input.Z) + Vector3.new(0, input.Y, 0)) * flySpeed
        end

        bv.Velocity = bv.Velocity:Lerp(desiredVelocity, 0.25)

        local lookXZ = Vector3.new(camera.CFrame.LookVector.X, 0, camera.CFrame.LookVector.Z)
        if lookXZ.Magnitude > 0.001 then
            local targetCF = CFrame.lookAt(root.Position, root.Position + lookXZ.Unit)
            bg.CFrame = bg.CFrame:Lerp(targetCF, 0.2)
        end

        local speedMag = bv.Velocity.Magnitude
        local alpha = math.clamp(speedMag / (flySpeed * 1.2), 0, 1)
        local targetFOV = defaultFOV + (maxFOV - defaultFOV) * alpha
        camera.FieldOfView = camera.FieldOfView + (targetFOV - camera.FieldOfView) * fovSmooth
    end)
end

local function stopFly()
    if not flying then return end
    flying = false
    if flyConnection then
        flyConnection:Disconnect()
        flyConnection = nil
    end
    local char = player.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            for _, obj in ipairs(root:GetChildren()) do
                if obj:IsA("BodyGyro") or obj:IsA("BodyVelocity") then
                    obj:Destroy()
                end
            end
        end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.PlatformStand = false
            hum.AutoRotate = true
        end
    end
    if camera then
        camera.FieldOfView = 70
    end
end

SttTab:Toggle({
    Title = "Enable Fly",
    Default = false,
    Callback = function(state)
        if state then
            startFly()
        else
            stopFly()
        end
    end
})

    end
