            local p = head.Position:Lerp(hrp.Position, 0.4)
            return Vector3.new(p.X, p.Y + CAM_Y_OFFSET, p.Z)
        end
        return (head and head.Position) or (hrp and hrp.Position)
    end

    local function findTargetByFOV()
        local cam = workspace.CurrentCamera
        if not cam then return nil end
        local center = Vector2.new(
            cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5)
        local bestD, bestPlr = math.huge, nil
        for _, plr in ipairs(Players:GetPlayers()) do
            local char = getAlive(plr)
            if char then
                local part = char:FindFirstChild("Head")
                    or char:FindFirstChild("HumanoidRootPart")
                if part then
                    local sp, onScreen = cam:WorldToViewportPoint(part.Position)
                    if onScreen then
                        local d = (center - Vector2.new(sp.X, sp.Y)).Magnitude
                        if d < bestD and d <= FOV_PIXELS then
                            bestD, bestPlr = d, plr
                        end
                    end
                end
            end
        end
        return bestPlr
    end

    local function findTargetByMouse()
        local origin = Mouse.Hit and Mouse.Hit.Position
        if not origin then return nil end
        local bestD, bestPlr = math.huge, nil
        for _, plr in ipairs(Players:GetPlayers()) do
            local char = getAlive(plr)
            if char then
                local hrp = char:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local d = (origin - hrp.Position).Magnitude
                    if d < bestD and d <= CAMLOCK_RADIUS then
                        bestD, bestPlr = d, plr
                    end
                end
            end
        end
        return bestPlr
    end

    local function updateVisuals(target, aimPos)
        ensureHighlight()
        if target and target.Character then
            highlight.Adornee = target.Character
            highlight.Enabled = true
        else
            highlight.Enabled = false
            highlight.Adornee = nil
        end

        ensureLine()
        if line and aimPos and (SilentAimEnabled or CamLockEnabled) then

    end

    local function onUpdate()
        if not SilentAimEnabled and not CamLockEnabled then
            clearVisuals()
            return
        end

        local now = tick()
        local dt  = math.clamp(now - lastFrameTime, 0.001, 0.1)
        lastFrameTime = now

        local cam = workspace.CurrentCamera
        if not cam then return end

        if not isUsable(ActiveTarget) then
            ActiveTarget = resolveTarget()
        end

        local char = getAlive(ActiveTarget)
        if not char then
            local didGrace = false

            if SilentAimEnabled and LastSilentPos
                and (now - LastSilentTime) < GRACE_TIME
            then
                local myHRP = getMyHRP()
                if myHRP then
                    local flat = Vector3.new(
                        LastSilentPos.X - myHRP.Position.X, 0,
                        LastSilentPos.Z - myHRP.Position.Z)
                    if flat.Magnitude > MIN_FLAT_DIST then
                        local cf = CFrame.new(myHRP.Position,
                            myHRP.Position + flat.Unit)
                        myHRP.CFrame = myHRP.CFrame:Lerp(cf, ROT_LERP_ALPHA)
                    end
                end
                didGrace = true
            end

            if CamLockEnabled and LastCamPos
                and (now - LastCamTime) < GRACE_TIME
            then
                local cur = cam.CFrame
                if (LastCamPos - cur.Position).Magnitude > MIN_CAM_DIST then
                    cam.CFrame = cur:Lerp(
                        CFrame.new(cur.Position, LastCamPos), CAM_MIN_ALPHA)
                end
                didGrace = true
            end

            if didGrace then
                updateVisuals(ActiveTarget, LastCamPos or LastSilentPos)
                return
            end

            ActiveTarget    = nil
            LastSilentPos   = nil
            LastCamPos      = nil
            clearVisuals()
            return
        end

        local pred = getPrediction()

        if SilentAimEnabled then
            local part = getSilentAimPart(ActiveTarget)
            if part then
                local aimPos = part.Position
                local myHRP  = getMyHRP()
                if myHRP then
                    local flat = Vector3.new(
                        aimPos.X - myHRP.Position.X, 0,
                        aimPos.Z - myHRP.Position.Z)
                    if flat.Magnitude > MIN_FLAT_DIST then
                        local cf = CFrame.new(myHRP.Position,
                            myHRP.Position + flat.Unit)
                        myHRP.CFrame = myHRP.CFrame:Lerp(cf, ROT_LERP_ALPHA)
                    end
                end
                LastSilentPos, LastSilentTime = aimPos, now
            end
        end

        if CamLockEnabled then
            local basePos = getCamAimPos(char)
            if basePos then
                local hrp    = char:FindFirstChild("HumanoidRootPart")
                local aimPos = hrp
                    and (basePos + hrp.AssemblyLinearVelocity * pred)
                    or basePos

                local cur = cam.CFrame
                local dir = aimPos - cur.Position
                local dist = dir.Magnitude

                if dist > MIN_CAM_DIST then
                    local dot   = math.clamp(cur.LookVector:Dot(dir.Unit), -1, 1)
                    local angle = math.acos(dot)
                    local base  = CAM_MIN_ALPHA + (CAM_MAX_ALPHA - CAM_MIN_ALPHA)
                                  * math.clamp(angle / math.pi, 0, 1)
                    local alpha = math.clamp(base * (dt / 0.016), 0.05, 0.7)
                    cam.CFrame = cur:Lerp(CFrame.new(cur.Position, aimPos), alpha)
                end
                LastCamPos, LastCamTime = aimPos, now
            end
        end

        updateVisuals(ActiveTarget, LastCamPos or LastSilentPos)
    end

    local function updateConnection()
        local shouldRun = SilentAimEnabled or CamLockEnabled
        if shouldRun then
            if not connUpdate then
                lastFrameTime = tick()

                    if not ok then warn("[AIM] onUpdate:", err) end
                end)
            end
        else
            if connUpdate then connUpdate:Disconnect(); connUpdate = nil end
            ActiveTarget, LastSilentPos, LastCamPos = nil, nil, nil
            clearVisuals()
        end
    end

function OrbitController:Start(plr)
    if not plr or plr == LocalPlayer then return false end
    if self.Enabled then self:Stop() end

    local localHRP = GetHRP(LocalPlayer)
    if not localHRP then
        local character = LocalPlayer.Character
        if character then
            character:WaitForChild("Humanoid", 5)
            localHRP = character:WaitForChild("HumanoidRootPart", 5)
        end
    end

    local targetHRP = GetHRP(plr)
    if not localHRP or not targetHRP then
        warn("[Orbit] Không tìm thấy HRP")
        return false
    end

    self.Enabled = true
    self.Target = plr

    local offsetVec = localHRP.Position - targetHRP.Position
    if offsetVec.Magnitude > 0.01 then
        self.Angle = math.atan2(offsetVec.Z, offsetVec.X)
    else
        self.Angle = 0
    end

    if self.CameraPart then self.CameraPart:Destroy() end
    self.CameraPart = Instance.new("Part")
    self.CameraPart.Name = "OrbitCameraFocus"
    self.CameraPart.Size = Vector3.new(0.1, 0.1, 0.1)
    self.CameraPart.Transparency = 1
    self.CameraPart.CanCollide = false
    self.CameraPart.Anchored = true
    self.CameraPart.CFrame = CFrame.new(targetHRP.Position)
    self.CameraPart.Parent = Workspace

    local cam = Workspace.CurrentCamera
    if cam then
        cam.CameraType = Enum.CameraType.Custom
        cam.CameraSubject = self.CameraPart
    end

    local targetPos = targetHRP.Position
    local offset = Vector3.new(math.cos(self.Angle) * Radius, Height, math.sin(self.Angle) * Radius)
    local startPos = targetPos + offset
    localHRP.CFrame = CFrame.lookAt(startPos, targetPos)
    localHRP.AssemblyLinearVelocity = Vector3.zero
    localHRP.AssemblyAngularVelocity = Vector3.zero

    self:BeginOrbitLoop(plr)
    return true
end

function OrbitController:BeginOrbitLoop(plr)
    if not self.Enabled then return end

    if self.CameraConnection then self.CameraConnection:Disconnect(); self.CameraConnection = nil end
    if self.PhysicsConnection then self.PhysicsConnection:Disconnect(); self.PhysicsConnection = nil end
    if self.LineConnection then self.LineConnection:Disconnect(); self.LineConnection = nil end

    self.CameraConnection = RunService.RenderStepped:Connect(function(dt)
        if not self.Enabled then return end
        local target = self.Target
        if not target then return end
        local targetRoot = GetHRP(target)
        if not targetRoot then return end

        self.Angle = (self.Angle + AngleSpeed * dt) % (math.pi * 2)

        local targetPosition = targetRoot.Position
        local offset = Vector3.new(math.cos(self.Angle) * Radius, Height, math.sin(self.Angle) * Radius)
        local desiredPosition = targetPosition + offset

        OrbitState.TargetPosition = desiredPosition
        OrbitState.TargetLookAt = targetPosition

        if self.CameraPart then
            self.CameraPart.CFrame = CFrame.new(targetPosition)
        end
    end)

    self.PhysicsConnection = RunService.Heartbeat:Connect(function()
        if not self.Enabled then return end
        local myRoot = GetHRP(LocalPlayer)
        local targetRoot = GetHRP(self.Target)
        if not myRoot or not targetRoot then return end

        local desiredPosition = targetRoot.Position + Vector3.new(
            math.cos(self.Angle) * Radius,
            Height,
            math.sin(self.Angle) * Radius
        )
        local lookAt = targetRoot.Position

        myRoot.CFrame = CFrame.lookAt(desiredPosition, lookAt)

        myRoot.AssemblyLinearVelocity = Vector3.zero
        myRoot.AssemblyAngularVelocity = Vector3.zero

        if RemoteEvent then
            RemoteEvent:FireServer(nil, myRoot.CFrame)
        end
    end)

    if not self.Line then
        self.Line = Drawing.new("Line")
        self.Line.Thickness = 2
        self.Line.Color = Color3.fromRGB(255, 0, 0)
        self.Line.Transparency = 1
        self.Line.Visible = false
    end

    self.LineConnection = RunService.RenderStepped:Connect(function()
        if not self.Enabled or not self.Target then
            self.Line.Visible = false
            return
        end
        local targetRoot = GetHRP(self.Target)
        local myRoot = GetHRP(LocalPlayer)
        local cam = Workspace.CurrentCamera
        if not targetRoot or not myRoot or not cam then
            self.Line.Visible = false
            return
        end
        local myPos, on1 = cam:WorldToViewportPoint(myRoot.Position)
        local tgtPos, on2 = cam:WorldToViewportPoint(targetRoot.Position)
        if on1 and on2 then
            self.Line.Visible = true
            self.Line.From = Vector2.new(myPos.X, myPos.Y)
            self.Line.To = Vector2.new(tgtPos.X, tgtPos.Y)
        else
            self.Line.Visible = false
        end
    end)
end

function OrbitController:Stop()
    self.Enabled = false
    self.Target = nil

    if self.PhysicsConnection then
        self.PhysicsConnection:Disconnect()
        self.PhysicsConnection = nil
    end
    OrbitState.TargetPosition = nil
    OrbitState.TargetLookAt = nil

    if self.CameraConnection then self.CameraConnection:Disconnect(); self.CameraConnection = nil end
    if self.LineConnection then self.LineConnection:Disconnect(); self.LineConnection = nil end
    if self.Line then self.Line.Visible = false end
    if self.CameraPart then self.CameraPart:Destroy(); self.CameraPart = nil end

    local cam = Workspace.CurrentCamera
    if cam then
        local character = LocalPlayer.Character
        if character then
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            cam.CameraType = Enum.CameraType.Custom
            if humanoid then cam.CameraSubject = humanoid end
        else
            cam.CameraSubject = nil
        end
    end
    LocalPlayer.DevCameraOcclusionMode = defaultOcclusion
end

Tab:Section({

do
    local Players     = game:GetService("Players")
    local RunService  = game:GetService("RunService")
    local Workspace   = game:GetService("Workspace")
    local LocalPlayer = Players.LocalPlayer

    local CFG = {
        Enabled              = false,
        AttackRange          = 9,
        ApproachSlack        = 3,
        CloseRange           = 5,
        FacingThreshold      = 0.7,
        CloseFacingThreshold = 0.35,
        TrackRange           = 45,
        TrackRangeBuffer     = 1.5,
        AttackAnimIds        = {},
        MinWindow            = 0.30,
        MaxWindow            = 0.80,
        DefaultWindow        = 0.45,
        RehitDelay           = 0.25,
        UsePrediction        = true,
        MaxLead              = 0.22,
        ExtraLead            = 0.016,
        SmoothRate           = 45,
        MaxSaneSpeed         = 220,
        StaleSampleGap       = 0.5,
        AnimDedupWindow      = 0.05,
        AnimatorWaitTime     = 5,
        TPBehindDistance     = 3,
        Cooldown             = 0.12,
        FailRetryDelay       = 0.03,
        AntiFeint            = true,
        PlayersOnly          = false,
        WallCheck            = true,
        Debug                = false,
        ProximityBoostRange  = 8,
        ProximityWindowMult  = 1.35,
        UseInfoDetection     = true,
        InfoFolderName       = "Info",
        CharactersFolderName = "Characters",
    }

    local AttackStateNames = {
        "Attacking","IsAttacking","Attack","Combat","Hitting","Hit",
        "Swinging","Swing","Punching","Punch","Slashing","Slash",
        "Casting","SkillActive","InCombo","ComboActive",
    }

    local MovementPriorities = {
        [Enum.AnimationPriority.Core]     = true,
        [Enum.AnimationPriority.Idle]     = true,
        [Enum.AnimationPriority.Movement] = true,
    }

    local ActionPriorities = {
        [Enum.AnimationPriority.Action]  = true,
        [Enum.AnimationPriority.Action2] = true,
        [Enum.AnimationPriority.Action3] = true,
        [Enum.AnimationPriority.Action4] = true,
    }

    local MovementKeywords = {
        "run","walk","idle","jump","fall","dash","climb","swim",
        "emote","spawn","sit","land","getup","ragdoll","hover",
        "stun","block","move","strafe","crouch","prone","roll",
    }

    local PI, TAU = math.pi, math.pi * 2

    local lastTP      = 0
    local attackers   = {}
    local tracker     = {}
    local hooks       = {}
    local globalConns = {}
    local animSeen    = setmetatable({}, { __mode = "k" })
    local infoCache   = setmetatable({}, { __mode = "k" })

    local function dbg(...) if CFG.Debug then print("[TPBehind]", ...) end end

    local function getLocal()
        local char = LocalPlayer.Character
        if not char then return nil, nil end
        return char:FindFirstChild("HumanoidRootPart"), char
    end

    local function attrActive(v)
        local t = type(v)
        if t == "nil"     then return false end
        if t == "boolean" then return v end
        if t == "number"  then return v > 0 end
        if t == "string"  then
            local l = string.lower(v)
            return l ~= "" and l ~= "false" and l ~= "0" and l ~= "no"
        end
        return false
    end

    local function isAlive(char)
        if not char or not char.Parent then return false end
        if attrActive(char:GetAttribute("Dead"))    then return false end
        if attrActive(char:GetAttribute("Ragdoll")) then return false end
        local hum = char:FindFirstChildOfClass("Humanoid")
        return hum ~= nil and hum.Health > 0
    end

    local function getInfoFolder(char)
        if not CFG.UseInfoDetection then return nil end

        local cached = infoCache[char]
        if cached then
            if cached.Parent then return cached end
            infoCache[char] = nil
        elseif cached == false then
            return nil
        end

        local info = char:FindFirstChild(CFG.InfoFolderName)
        if info then
            infoCache[char] = info
            return info
        end

        local charsFolder = Workspace:FindFirstChild(CFG.CharactersFolderName)
        if charsFolder then
            local model = charsFolder:FindFirstChild(char.Name)
            if model then
                info = model:FindFirstChild(CFG.InfoFolderName)
                if info then
                    infoCache[char] = info
                    return info
                end
            end
        end

        infoCache[char] = false
        return nil
    end

    local function invalidateInfoCache(char)
        infoCache[char] = nil
    end

    local function isCharacterAttacking(char)
        local info = getInfoFolder(char)

        if not info then return nil end

        local foundFlag = false

        for _, name in ipairs(AttackStateNames) do
            local child = info:FindFirstChild(name)
            if child then
                foundFlag = true
                local val
                if child:IsA("BoolValue") then
                    val = child.Value
                elseif child:IsA("StringValue") then
                    val = attrActive(child.Value)
                elseif child:IsA("IntValue") or child:IsA("NumberValue") then
                    val = child.Value > 0
                end
                if val then return true end
            end

            local attr = info:GetAttribute(name)
            if attr ~= nil then
                foundFlag = true
                if attrActive(attr) then return true end
            end
        end

        for _, name in ipairs(AttackStateNames) do
            local attr = char:GetAttribute(name)
            if attr ~= nil then
                foundFlag = true
                if attrActive(attr) then return true end
            end
        end

        if not foundFlag then return nil end

        return false
    end

    local function extractAnimId(track)
        local anim = track.Animation
        if not anim then return nil end
        local id = anim.AnimationId or ""
        local s = string.match(id, "(%d+)")
        return s and tonumber(s) or nil
    end

    local function isMovementAnim(track)
        if MovementPriorities[track.Priority] then return true end

        if next(CFG.AttackAnimIds) then
            local id = extractAnimId(track)
            if id and not CFG.AttackAnimIds[id] then return true end
        end

        local anim = track.Animation
        local n = anim and string.lower(anim.Name or "") or ""
        if n ~= "" then
            for _, kw in ipairs(MovementKeywords) do
                if string.find(n, kw, 1, true) then return true end
            end
        end

        return false
    end

    local function isAttackAnimFallback(track)
        if not track then return false end
        if isMovementAnim(track) then return false end

        if ActionPriorities[track.Priority] then return true end

        local anim = track.Animation
        local n = anim and (anim.Name or "") or ""
        return n ~= ""
    end

    local function animDedup(track)
        local now = os.clock()
        local last = animSeen[track]
        if last and (now - last) < CFG.AnimDedupWindow then return true end
        animSeen[track] = now
        return false
    end

    local function flatLook(cf)
        local l = cf.LookVector
        local f = Vector3.new(l.X, 0, l.Z)
        if f.Magnitude < 0.01 then return Vector3.new(0, 0, -1) end
        return f.Unit
    end

    local function yawOf(look)   return math.atan2(look.X, look.Z) end
    local function yawToDir(yaw) return Vector3.new(math.sin(yaw), 0, math.cos(yaw)) end

    local function getLead()
        local ping = 0
        pcall(function() ping = LocalPlayer:GetNetworkPing() end)
        return math.clamp(ping + CFG.ExtraLead, 0, CFG.MaxLead)
    end

    local function sample(char, root, now)
        local pos  = root.Position
        local look = flatLook(root.CFrame)
        local yaw  = yawOf(look)
        local s    = tracker[char]

        if not s then
            tracker[char] = { pos = pos, yaw = yaw, vel = Vector3.zero, w = 0, t = now }
            return
        end

        local dt = now - s.t
        if dt <= 1e-4 then return end

        if dt > CFG.StaleSampleGap then
            tracker[char] = { pos = pos, yaw = yaw, vel = Vector3.zero, w = 0, t = now }
            return
        end

        local rawVel = (pos - s.pos) / dt
        if rawVel.Magnitude > CFG.MaxSaneSpeed then
            tracker[char] = { pos = pos, yaw = yaw, vel = Vector3.zero, w = 0, t = now }
            return
        end

        local dy   = ((yaw - s.yaw + PI) % TAU) - PI
        local rawW = math.clamp(dy / dt, -40, 40)
        local a    = 1 - math.exp(-dt * CFG.SmoothRate)

        s.vel = s.vel:Lerp(rawVel, a)
        s.w   = s.w + (rawW - s.w) * a
        s.pos, s.yaw, s.t = pos, yaw, now
    end

    local function isEnemyFeinting(char)
        if not CFG.AntiFeint then return false end
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return false end
        local feint = root:FindFirstChild("Feint")
        if not feint then return false end
        local ring = feint:FindFirstChild("Ring")
        if ring and ring:IsA("ParticleEmitter") then
            for _, kp in ipairs(ring.Color.Keypoints) do
                local v = kp.Value
                if math.abs(v.R - v.G) < 0.05 and math.abs(v.G - v.B) < 0.05 then
                    return true
                end
            end
        end
        return false
    end

    local function dotAt(look, ePos, mPos, dist)
        local dir = Vector3.new(mPos.X - ePos.X, 0, mPos.Z - ePos.Z)
        if dir.Magnitude < 0.1 then return 1 end
        local threshold = dist <= CFG.CloseRange
            and CFG.CloseFacingThreshold
            or  CFG.FacingThreshold
        local d = look:Dot(dir.Unit)
        return d >= threshold and d or nil
    end

    local function scoreCandidate(char, root, localRoot, lead)
        local s = tracker[char]

        local curPos  = root.Position
        local curDist = (localRoot.Position - curPos).Magnitude
        if curDist > CFG.AttackRange + CFG.ApproachSlack then return nil end

        local curLook  = flatLook(root.CFrame)
        local predPos  = curPos
        local predLook = curLook
        local myPos    = localRoot.Position

        if CFG.UsePrediction and s then
            predPos  = curPos + s.vel * lead
            predLook = yawToDir(s.yaw + s.w * lead)
            myPos    = localRoot.Position + localRoot.AssemblyLinearVelocity * lead
        end

        local predDist = (myPos - predPos).Magnitude
        if math.min(curDist, predDist) > CFG.AttackRange then return nil end

        local d1  = dotAt(curLook, curPos, localRoot.Position, curDist)
        local d2  = dotAt(predLook, predPos, myPos, predDist)
        local dot = (d1 and d2 and math.max(d1, d2)) or d1 or d2
        if not dot then return nil end

        local closing = 0
        if s then
            local toMe = localRoot.Position - curPos
            if toMe.Magnitude > 0.1 then
                closing = math.max(0, s.vel:Dot(toMe.Unit))
            end
        end

        return math.min(curDist, predDist) * (2 - dot) - closing * 0.05
    end

    local function tpBehind(enemyChar, enemyRoot)
        local localRoot, localChar = getLocal()
        if not localRoot or not isAlive(localChar) then return false end

        local s        = tracker[enemyChar]
        local enemyPos = enemyRoot.Position
        local look     = flatLook(enemyRoot.CFrame)

        if CFG.UsePrediction and s then
            local lead = getLead() * 0.5
            enemyPos = enemyPos + s.vel * lead
            look     = yawToDir(s.yaw + s.w * lead)
        end

        local back = -look
        local dist = CFG.TPBehindDistance

        if CFG.WallCheck then
            local params = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Exclude
            params.FilterDescendantsInstances = { localChar, enemyChar }
            local hit = Workspace:Raycast(enemyPos, back * dist, params)
            if hit and hit.Instance.CanCollide then
                dist = math.max(1.5, (hit.Position - enemyPos).Magnitude - 1)
            end
        end

        local behindPos = enemyPos + back * dist
        local lookAtPos = Vector3.new(enemyPos.X, behindPos.Y, enemyPos.Z)

        local ok = pcall(function()
            localChar:PivotTo(CFrame.lookAt(behindPos, lookAtPos))
        end)

        lastTP = ok
            and os.clock()
            or  os.clock() - CFG.Cooldown + CFG.FailRetryDelay

        dbg("TP", enemyChar.Name, ok)
        return ok
    end

    local function evaluate()
        if not CFG.Enabled then return end

        local now = os.clock()
        if now - lastTP < CFG.Cooldown then return end

        local localRoot, localChar = getLocal()
        if not localRoot or not isAlive(localChar) then return end

        local lead = getLead()
        local bestChar, bestRoot, bestScore

        for char, data in pairs(attackers) do
            if now > data.expire or not isAlive(char) then
                attackers[char] = nil
            elseif not data.handled then
                local root = char:FindFirstChild("HumanoidRootPart")
                if root and not isEnemyFeinting(char) then
                    local score = scoreCandidate(char, root, localRoot, lead)
                    if score and (not bestScore or score < bestScore) then
                        bestScore, bestChar, bestRoot = score, char, root
                    end
                end
            end
        end

        if bestChar then
            local ok = tpBehind(bestChar, bestRoot)
            local d  = attackers[bestChar]
            if d then d.handled = true; d.handledAt = now end
            dbg("→", bestChar.Name, ok)
        end
    end

    local function onAnimationPlayed(char, track)
        if not CFG.Enabled then return end
        if char == LocalPlayer.Character then return end
        if animDedup(track) then return end

        local state = isCharacterAttacking(char)

        if state == false then
            dbg("Info: not attacking →", char.Name)
            return
        end

        if state == nil then
            if not isAttackAnimFallback(track) then
                dbg("Anim: movement skip →", char.Name,
                    track.Animation and track.Animation.Name or "(noname)",
                    tostring(track.Priority))
                return
            end
        end

        if not isAlive(char) then return end

        local len = track.Length
        local window
        if len and len > 0 then
            window = math.min(len, CFG.MaxWindow)
        else
            window = math.clamp(CFG.DefaultWindow, CFG.MinWindow, CFG.MaxWindow)
        end

        local localRoot = getLocal()
        if localRoot then
            local root = char:FindFirstChild("HumanoidRootPart")
            if root then
                local dist = (localRoot.Position - root.Position).Magnitude
                if dist <= CFG.ProximityBoostRange then
                    window = math.min(window * CFG.ProximityWindowMult, CFG.MaxWindow * 1.5)
                    dbg("Proximity boost →", char.Name, "d:", dist, "w:", window)
                end
            end
        end

        local now = os.clock()
        local old = attackers[char]
        if old and old.handled and (now - (old.handledAt or 0)) < CFG.RehitDelay then
            old.expire = math.max(old.expire, now + window)
        else
            attackers[char] = { expire = now + window, handled = false, handledAt = 0 }
        end

        evaluate()
    end

    local function unhook(char)
        local conns = hooks[char]
        if conns then
            for i = 1, #conns do
                pcall(function() conns[i]:Disconnect() end)
            end
        end
        hooks[char]     = nil
        attackers[char] = nil
        tracker[char]   = nil
        infoCache[char] = nil
    end

    local function hookCharacter(char)
        if hooks[char] then return end
        if char == LocalPlayer.Character then return end
        if not char:IsA("Model") then return end
        if CFG.PlayersOnly and not Players:GetPlayerFromCharacter(char) then return end

        local conns = {}
        hooks[char] = conns

        local hum = char:WaitForChild("Humanoid", 5)
        if not hum or hooks[char] ~= conns then unhook(char); return end

        local animator = hum:FindFirstChildOfClass("Animator")
            or hum:WaitForChild("Animator", CFG.AnimatorWaitTime)
        if hooks[char] ~= conns then return end

        local source = animator or hum

        conns[#conns + 1] = source.AnimationPlayed:Connect(function(track)
            onAnimationPlayed(char, track)
        end)

        conns[#conns + 1] = char.AncestryChanged:Connect(function(_, parent)
            if not parent then unhook(char) end
        end)

        conns[#conns + 1] = char:GetPropertyChangedSignal("Name"):Connect(function()
            invalidateInfoCache(char)
        end)

        dbg("hooked", char.Name, animator and "(Animator)" or "(Hum fallback)")
    end

    local function onFrame()
        if not CFG.Enabled then return end

        local localRoot = getLocal()
        if not localRoot then return end

        local now    = os.clock()
        local range2 = CFG.TrackRange * CFG.TrackRange
        local drop2  = (CFG.TrackRange * CFG.TrackRangeBuffer) ^ 2

        for char in pairs(hooks) do
            local root = char:FindFirstChild("HumanoidRootPart")
            if root then
                local dv = root.Position - localRoot.Position
                local d2 = dv:Dot(dv)
                if d2 <= range2 then
                    sample(char, root, now)
                elseif d2 > drop2 then
                    tracker[char] = nil
                end
            end
        end

        if next(attackers) then evaluate() end
    end

    local function hookPlayer(plr)
        if plr == LocalPlayer then return end
        if plr.Character then task.spawn(hookCharacter, plr.Character) end
        globalConns[#globalConns + 1] = plr.CharacterAdded:Connect(function(c)
            task.wait(0.2)
            hookCharacter(c)
        end)
    end

    local function watchCharsFolder(folder)
        for _, c in ipairs(folder:GetChildren()) do
            task.spawn(hookCharacter, c)
        end
        globalConns[#globalConns + 1] = folder.ChildAdded:Connect(function(c)
            task.wait(0.3)
            hookCharacter(c)
        end)
    end

    local function setEnabled(v)
        v = v and true or false
        if CFG.Enabled == v then return end
        CFG.Enabled = v
        if not v then
            table.clear(attackers)
            table.clear(tracker)
            lastTP = 0
        end
        print("[TPBehind] Enabled =", v)
    end

    _G.TPBehindSetEnabled = setEnabled
    _G.TPBehindIsEnabled  = function() return CFG.Enabled end

    local function init()
        for _, plr in ipairs(Players:GetPlayers()) do hookPlayer(plr) end
        globalConns[#globalConns + 1] = Players.PlayerAdded:Connect(hookPlayer)

        if not CFG.PlayersOnly then
            local folder = Workspace:FindFirstChild(CFG.CharactersFolderName)
            if folder then
                watchCharsFolder(folder)
            else
                globalConns[#globalConns + 1] = Workspace.ChildAdded:Connect(function(c)
                    if c.Name == CFG.CharactersFolderName then
                        watchCharsFolder(c)
                    end
                end)
            end
        end

        globalConns[#globalConns + 1] = RunService.PreSimulation:Connect(onFrame)

        globalConns[#globalConns + 1] = LocalPlayer.CharacterAdded:Connect(function()
            table.clear(attackers)
            table.clear(tracker)
            lastTP = 0
        end)

        _G.TPBehindStop = function()
            for _, c in ipairs(globalConns) do
                pcall(function() c:Disconnect() end)
            end
            for char in pairs(hooks) do unhook(char) end
            table.clear(globalConns)
            table.clear(attackers)
            table.clear(tracker)
            _G.TPBehindStop       = nil
            _G.TPBehindSetEnabled = nil
            _G.TPBehindIsEnabled  = nil
            print("[TPBehind] Stopped.")
        end
    end

    init()

    Tab:Toggle({
        Title    = "skill moves",
        Value    = CFG.Enabled,
        Callback = function(value)
            if _G.TPBehindSetEnabled then
                _G.TPBehindSetEnabled(value)
            end
        end,
    })

local function ValidateStable(character)
    local humanoid = GetHumanoid(character)
    local root = GetRoot(character)
    if not humanoid or not root then
        return false
    end
    local startTime = os.clock()
    while os.clock() - startTime < CONFIG.StableTime do
        if not State.Enabled then
            return false
        end
        if Player.Character ~= character then
            return false
        end
        if humanoid.Health <= 0 then
            return false
        end
        if not IsAtTarget(root) then
            return false
        end
        task.wait(CONFIG.CheckInterval)
    end
    if Player.Character ~= character then
        return false
    end
    if humanoid.Health <= 0 then
        return false
    end
    return IsAtTarget(root)

local function ExecuteCycle()
    State.Cycle += 1
    local oldCharacter = GetCharacter()
    local oldHumanoid = GetHumanoid(oldCharacter)
    if not oldCharacter or not oldHumanoid or oldHumanoid.Health <= 0 then
        return false
    end
    if not FireReset() then
        return false
    end
    local newCharacter = WaitForNewCharacter(oldCharacter)
    if not newCharacter then
        return false
    end
    local delayStart = os.clock()
    while os.clock() - delayStart < CONFIG.RespawnDelay do
        if not State.Enabled then
            return false
        end
        local humanoid = GetHumanoid(newCharacter)
        if not humanoid or humanoid.Health <= 0 then
            return false
        end
        task.wait(0.05)
    end
    if not TweenToTarget(newCharacter) then
        return false
    end
    local root = GetRoot(newCharacter)
    if not root then
        return false
    end
    if not IsAtTarget(root) then
        return false
    end
    return ValidateStable(newCharacter)
end

                break
            end
            if State.Enabled then
                task.wait(CONFIG.RetryDelay)
            end
        end
        State.Running = false
    end)
end

        local center = joystickBase.AbsolutePosition + joystickBase.AbsoluteSize / 2
        local position = Vector2.new(input.Position.X, input.Position.Y)
        local delta = position - center
        if delta.Magnitude > JOYSTICK_RADIUS then
            delta = delta.Unit * JOYSTICK_RADIUS
        end
        knob.Position = UDim2.new(0.5, delta.X, 0.5, delta.Y)
        CameraSystem.MoveInput = Vector2.new(
            delta.X / JOYSTICK_RADIUS,
            delta.Y / JOYSTICK_RADIUS
        )
    end

    local function ResetJoystick()
        joystickActive = false
        joystickTouch = nil
        CameraSystem.MoveInput = Vector2.zero
        knob.Position = UDim2.fromScale(0.5, 0.5)
    end

    table.insert(CameraSystem.TouchConnections,
        joystickBase.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch then
                joystickActive = true
                joystickTouch = input
                UpdateJoystick(input)
            end
        end)
    )

    table.insert(CameraSystem.TouchConnections,
        UserInputService.InputChanged:Connect(function(input)
            if input == joystickTouch and joystickActive then
                UpdateJoystick(input)
            end
        end)
    )

    table.insert(CameraSystem.TouchConnections,
        UserInputService.TouchEnded:Connect(function(input)
            if input == joystickTouch then
                ResetJoystick()
            end
        end)
    )

    local function HoldButton(button, onState)
        table.insert(CameraSystem.TouchConnections,
            button.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.Touch
                    or input.UserInputType == Enum.UserInputType.MouseButton1 then
                    onState(true)
                end
            end)
        )
        table.insert(CameraSystem.TouchConnections,
            button.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.Touch
                    or input.UserInputType == Enum.UserInputType.MouseButton1 then
                    onState(false)
                end
            end)
        )
    end

    HoldButton(upButton, function(state)
        CameraSystem.UpPressed = state
    end)
    HoldButton(downButton, function(state)
        CameraSystem.DownPressed = state
    end)

    local activeLookTouch = nil
    local lastTouchPosition = nil

    local function IsTouchOnControl(pos)
        return IsPointInGui(pos, joystickBase)
            or IsPointInGui(pos, upButton)
            or IsPointInGui(pos, downButton)
    end

    table.insert(CameraSystem.TouchConnections,
        lookArea.InputBegan:Connect(function(input)
            if input.UserInputType ~= Enum.UserInputType.Touch then
                return
            end
            local pos = Vector2.new(input.Position.X, input.Position.Y)
            if IsTouchOnControl(pos) then
                return
            end
            activeLookTouch = input
            lastTouchPosition = input.Position
        end)
    )

    table.insert(CameraSystem.TouchConnections,
        lookArea.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch and input == activeLookTouch then
                if lastTouchPosition then
                    local delta = input.Position - lastTouchPosition
                    CameraSystem.Yaw -= delta.X * TOUCH_SENSITIVITY
                    CameraSystem.Pitch -= delta.Y * TOUCH_SENSITIVITY
                    ClampPitch()
                end
                lastTouchPosition = input.Position
            end
        end)
    )

    table.insert(CameraSystem.TouchConnections,
        UserInputService.TouchEnded:Connect(function(input)
            if input == activeLookTouch then
                activeLookTouch = nil
                lastTouchPosition = nil
            end
        end)
    )
end

local function EnableFreeCamera()
    if CameraSystem.Enabled then return end
    local Camera = GetCamera()
    local part = CreateTargetPart()
    if not Camera or not part then return end
    CameraSystem.Enabled = true
    CameraSystem.SavedCameraType = Camera.CameraType
    CameraSystem.SavedCameraSubject = Camera.CameraSubject
    CameraSystem.SavedCameraCFrame = Camera.CFrame
    CameraSystem.Position = part.Position
    CameraSystem.Yaw = START_YAW
    CameraSystem.Pitch = START_PITCH
    table.clear(CameraSystem.Keys)
    CameraSystem.UpPressed = false
    CameraSystem.DownPressed = false
    CameraSystem.MoveInput = Vector2.zero
    Camera.CameraType = Enum.CameraType.Scriptable

    CameraSystem.InputBegan = UserInputService.InputBegan:Connect(function(input, processed)
        if not CameraSystem.Enabled or processed then return end
        if input.UserInputType == Enum.UserInputType.Keyboard then
            CameraSystem.Keys[input.KeyCode] = true
        end
    end)

    CameraSystem.InputEnded = UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Keyboard then
            CameraSystem.Keys[input.KeyCode] = false
        end
    end)

    CameraSystem.InputChanged = UserInputService.InputChanged:Connect(function(input)
        if not CameraSystem.Enabled then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement then
            CameraSystem.Yaw -= input.Delta.X * LOOK_SENSITIVITY
            CameraSystem.Pitch -= input.Delta.Y * LOOK_SENSITIVITY
            ClampPitch()
        end
    end)

    CreateMobileControls()

    if not CameraSystem.RenderBound then
        RunService:BindToRenderStep(
            "KhanhlyFreeCamera",
            Enum.RenderPriority.Camera.Value + 1,
            function(deltaTime)
                if not CameraSystem.Enabled then return end
                local currentCamera = GetCamera()
                if not currentCamera then return end
                local rotation = GetCameraRotation()
                local movement = GetMovementDirection()
                if movement.Magnitude > 0 then
                    local worldDirection = rotation:VectorToWorldSpace(movement)
                    local speed = CameraSystem.Keys[Enum.KeyCode.LeftShift] and SPRINT_SPEED or MOVE_SPEED
                    CameraSystem.Position += worldDirection * speed * deltaTime
                end
                currentCamera.CameraType = Enum.CameraType.Scriptable
                currentCamera.CFrame = CFrame.new(CameraSystem.Position) * rotation
            end
        )
        CameraSystem.RenderBound = true
    end
end

local function DisableFreeCamera()
    if not CameraSystem.Enabled then return end
    CameraSystem.Enabled = false
    if CameraSystem.RenderBound then
        pcall(function()
            RunService:UnbindFromRenderStep("KhanhlyFreeCamera")
        end)
        CameraSystem.RenderBound = false
    end
    ClearConnections()
    table.clear(CameraSystem.Keys)
    CameraSystem.MoveInput = Vector2.zero
    CameraSystem.UpPressed = false
    CameraSystem.DownPressed = false
    if CameraSystem.MobileGui then
        CameraSystem.MobileGui:Destroy()
        CameraSystem.MobileGui = nil
    end
    local Camera = GetCamera()
    if Camera then
        local humanoid = GetCharacterHumanoid()
        Camera.CameraType = Enum.CameraType.Custom
        if humanoid then
            Camera.CameraSubject = humanoid
        end
    end
    CameraSystem.Position = nil
end

local function isDummyCharacter(character)
    if not character then return false end
    local container = Workspace:FindFirstChild("Characters")
    if not container then return false end
    local dummy = container:FindFirstChild("Dummy")
    if not dummy then return false end
    return character == dummy or character:IsDescendantOf(dummy)
end

local function getTargets()
    local targets = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            if not isDummyCharacter(plr.Character) then
                local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
                local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    table.insert(targets, hrp)
                end
            end
        end
    end
    return targets
end

local function findNearestTarget(fromPos)
    local nearest, minDist = nil, math.huge
    for _, target in ipairs(getTargets()) do
        if target and target.Parent then
            local dist = (target.Position - fromPos).Magnitude
            if dist < minDist then
                nearest, minDist = target, dist
            end
        end
    end
    return nearest, minDist
end

local function chooseTarget(mode, nameFilter)
    local targets = getTargets()
    if #targets == 0 then return nil end

    if mode == "Random" then
        return targets[math.random(1, #targets)]
    end

    if mode == "Nearest" then
        local myHRP = getHRP()
        if not myHRP then return nil end
        return findNearestTarget(myHRP.Position)
    end

    if mode == "Specified" and nameFilter and nameFilter ~= "" then
        local needle = nameFilter:lower()
        for _, target in ipairs(targets) do
            local plr  = Players:GetPlayerFromCharacter(target.Parent)
            local name = plr and plr.Name or target.Parent.Name
            if name:lower():find(needle, 1, true) then
                return target
            end
        end
    end
    return nil
end

local function runRoutine()
    local myChar = getMyChar()
    local tChar  = getTargetChar()
    local myHRP  = getHRP(myChar)
    local tHRP   = getHRP(tChar)

    if not (myHRP and tHRP) then return end

    local dir = (tHRP.Position - myHRP.Position)
    if dir.Magnitude > 0 then dir = dir.Unit else dir = Vector3.new(0, 0, 1) end
    local tpPos = tHRP.Position - dir * CONFIG.TPOffset

    myHRP.CFrame   = CFrame.new(tpPos, tHRP.Position)
    myHRP.Velocity = Vector3.zero
    task.wait(CONFIG.ScanDelay)

    local elapsed = 0
    while running do
        myChar = getMyChar()
        tChar  = getTargetChar()
        myHRP  = getHRP(myChar)
        tHRP   = getHRP(tChar)

        if not (myHRP and tHRP) then return end

        local dist = (myHRP.Position - tHRP.Position).Magnitude
        if dist <= CONFIG.ScanRange then
            break
        end

        dir   = (tHRP.Position - myHRP.Position).Unit
        tpPos = tHRP.Position - dir * CONFIG.TPOffset
        myHRP.CFrame   = CFrame.new(tpPos, tHRP.Position)
        myHRP.Velocity = Vector3.zero

        elapsed = elapsed + CONFIG.ScanDelay
        if elapsed >= CONFIG.ScanTimeout then
            return
        end
        task.wait(CONFIG.ScanDelay)
    end

    if not running then return end

    myChar = getMyChar()
    if not myChar then return end

    local moveset = myChar:FindFirstChild("Moveset")
    local skill   = moveset and moveset:FindFirstChild("Second Wind")
    if not skill then return end

    for i = 1, CONFIG.FireCount do
        if not running then return end
        pcall(function()
            SecondWindRemote:FireServer(skill, true)
        end)
        task.wait(CONFIG.FireDelay)
    end

    local waited = 0
    while waited < CONFIG.AfterFire do
        if not running then return end
        task.wait(0.05)
        waited = waited + 0.05
    end

    if not running then return end

    myChar = getMyChar()
    myHRP  = getHRP(myChar)
    if not myHRP then return end

    myHRP.CFrame   = CFrame.new(CONFIG.DropCoords)
    myHRP.Velocity = Vector3.zero
    lockFloat(myHRP)
end

local function startLoop()
    if running then return end
    running = true

    activeThread = task.spawn(function()
        while running do
            if selectedTarget then
                runRoutine()
            end
            task.wait(CONFIG.LoopDelay)
        end
    end)
end

local function stopLoop()
    running = false
    if activeThread then
        task.cancel(activeThread)
        activeThread = nil
    end
    unlockFloat()

    local myHRP = getHRP(getMyChar())
    if myHRP then
        myHRP.Velocity = Vector3.zero
    end
end

    if typeof(gethui) == "function" then
        local success, gui = pcall(gethui)

ESPGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

-- Get real character
local function GetRealCharacter(player)
    local folder = RefreshCharactersFolder()
    if not folder then return nil end

    local char = folder:FindFirstChild(player.Name)
    if not char or not char:IsA("Model") then return nil end
    if char.Parent ~= folder then return nil end

    local humanoid = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    local head = char:FindFirstChild("Head")
    if not humanoid or not root or not head then return nil end
    if humanoid.Health <= 0 then return nil end
    if root.Parent ~= char or head.Parent ~= char then return nil end

    return char
end

-- Ẩn toàn bộ ESP của một player (không xoá)
local function HideESP(data)
    if not data then return end
    if data.container then data.container.Visible = false end
    if data.highlight then data.highlight.Enabled = false end
    if data.snapLine then data.snapLine.Visible = false end
end

-- Xoá ESP khi player rời game
local function ClearPlayerESP(player)
    local data = EspData[player]
    if not data then return end

    if data.container then
        pcall(function() data.container:Destroy() end)
    end
    if data.highlight then
        pcall(function() data.highlight:Destroy() end)
    end
    if data.snapLine then
        pcall(function() data.snapLine:Remove() end)
    end
    EspData[player] = nil
end

local function ClearAllESP()
    for player in pairs(EspData) do
        ClearPlayerESP(player)
    end
end

-- Main update loop
local function UpdateESP()
    if not ESPEnabled then return end

    local folder = RefreshCharactersFolder()
    if not folder then return end

    local localChar = LocalPlayer.Character
    if not localChar then return end
    local localRoot = localChar:FindFirstChild("HumanoidRootPart")
    if not localRoot then return end

    local screenSize = Camera.ViewportSize
    local snapOrigin = Vector2.new(screenSize.X / 2, screenSize.Y - 2)

    local seenPlayers = {}

    for _, player in ipairs(Players:GetPlayers()) do
        seenPlayers[player] = true

        -- Bỏ qua chính mình
        if player == LocalPlayer then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        -- Lấy character thật
        local char = GetRealCharacter(player)
        if not char then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local rootPart = char:FindFirstChild("HumanoidRootPart")
        local head = char:FindFirstChild("Head")
        if not humanoid or not rootPart or not head then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        if humanoid.Health <= 0 then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        local distance = (rootPart.Position - localRoot.Position).Magnitude
        if distance > MAX_DISTANCE then
            if EspData[player] then HideESP(EspData[player]) end
            continue
        end

        -- Tạo/lấy data
        local data = EspData[player]
        if not data then
            data = CreatePlayerESP(player)
        end

        -- Hiện container
        data.container.Visible = true

        -- Highlight (chỉ đổi khi character thay đổi)
        if data.character ~= char then
            data.character = char
            data.highlight.Adornee = char
            data.highlight.Enabled = true
        elseif not data.highlight.Enabled then
            data.highlight.Enabled = true
        end

        -- SnapLine (Drawing.Line)
        local rootScreen = Camera:WorldToViewportPoint(rootPart.Position)
        if rootScreen.Z > 0 then
            local targetX = math.clamp(rootScreen.X, 0, screenSize.X)
            local targetY = math.clamp(rootScreen.Y, 0, screenSize.Y)
            data.snapLine.Visible = true
            data.snapLine.From = snapOrigin
            data.snapLine.To = Vector2.new(targetX, targetY)
        else
            data.snapLine.Visible = false
        end

        -- Tính toán Box/Health/Name/Distance
        local headPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
        local hipHeight = humanoid.HipHeight or 2
        local feetPos = Camera:WorldToViewportPoint(rootPart.Position - Vector3.new(0, hipHeight + 2, 0))

        if headPos.Z <= 0 and feetPos.Z <= 0 then
            data.box.Visible = false
            data.outline.Visible = false
            data.bgBar.Visible = false
            data.healthBar.Visible = false
            data.nameText.Visible = false
            data.distText.Visible = false
            continue
        end

        local top = math.min(headPos.Y, feetPos.Y)
        local bottom = math.max(headPos.Y, feetPos.Y)
        local height = math.max(bottom - top, 20)
        local width = math.max(height * 0.6, 24)
        local left = headPos.X - width / 2

        top = math.clamp(top, -50, screenSize.Y + 50)
        left = math.clamp(left, -50, screenSize.X + 50)

        -- Box
        data.box.Visible = true
        data.box.Position = UDim2.fromOffset(left, top)
        data.box.Size = UDim2.fromOffset(width, height)

        -- Outline
        data.outline.Visible = true
        data.outline.Position = UDim2.fromOffset(left - 2, top - 2)
        data.outline.Size = UDim2.fromOffset(width + 4, height + 4)

        -- Health bar
        local maxHealth = math.max(humanoid.MaxHealth, 1)
        local healthPercent = math.clamp(humanoid.Health / maxHealth, 0, 1)
        local barWidth = math.clamp(height / 25, 2, 4)
        local barLeft = left - barWidth - 6

        data.bgBar.Visible = true
        data.bgBar.Position = UDim2.fromOffset(barLeft, top)
        data.bgBar.Size = UDim2.fromOffset(barWidth, height)

        local fillHeight = height * healthPercent
        data.healthBar.Visible = true
        data.healthBar.Position = UDim2.fromOffset(barLeft, top + height - fillHeight)
        data.healthBar.Size = UDim2.fromOffset(barWidth, fillHeight)
        data.healthBar.BackgroundColor3 = Color3.fromRGB(
            255 * (1 - healthPercent),
            255 * healthPercent,
            0
        )

        -- Name
        data.nameText.Visible = true
        data.nameText.Text = player.Name
        data.nameText.Position = UDim2.fromOffset(headPos.X, top - 2)

        -- Distance
        data.distText.Visible = true
        data.distText.Text = string.format("M:%03d", math.floor(distance))
        data.distText.Position = UDim2.fromOffset(headPos.X, top + 16)
    end

    -- Xoá player đã rời
    for player in pairs(EspData) do
        if not seenPlayers[player] then
            ClearPlayerESP(player)
        end
    end
end

local player = game.Players.LocalPlayer
local camera = workspace.CurrentCamera
local runService = game:GetService("RunService")
local uis = game:GetService("UserInputService")

local flying = false
local flySpeed = 50
local flyConnection = nil
local controls = nil

pcall(function()
    local playerModule = require(player:WaitForChild("PlayerScripts"):WaitForChild("PlayerModule"))
    controls = playerModule:GetControls()
end)

local function getMoveVector()
    if controls then
        return controls:GetMoveVector()
    end
    local move = Vector3.zero
    if uis:IsKeyDown(Enum.KeyCode.W) then move += Vector3.new(0, 0, -1) end
    if uis:IsKeyDown(Enum.KeyCode.S) then move += Vector3.new(0, 0, 1) end
    if uis:IsKeyDown(Enum.KeyCode.A) then move += Vector3.new(-1, 0, 0) end
    if uis:IsKeyDown(Enum.KeyCode.D) then move += Vector3.new(1, 0, 0) end
    if uis:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
    if uis:IsKeyDown(Enum.KeyCode.LeftShift) then move += Vector3.new(0, -1, 0) end
    return move
end

local function startFly()
    if flying then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    flying = true

    hum.AutoRotate = false
    hum.PlatformStand = true

    local bg = Instance.new("BodyGyro")
    bg.P = 9e4
    bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bg.Parent = root

    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.Velocity = Vector3.zero
    bv.Parent = root

    local defaultFOV = camera.FieldOfView
    local maxFOV = 105
    local fovSmooth = 0.1

    flyConnection = runService.Heartbeat:Connect(function()
        if not flying or not char.Parent or not root.Parent or not hum.Parent then
            stopFly()
            return
        end

        local input = getMoveVector()
        local desiredVelocity

        if controls then
            local camRelative = Vector3.new(input.X, input.Y, input.Z)
            desiredVelocity = camera.CFrame:VectorToWorldSpace(camRelative) * flySpeed
        else
            local camLook = camera.CFrame.LookVector
            local camRight = camera.CFrame.RightVector
            local forward = Vector3.new(camLook.X, 0, camLook.Z).Unit
            local right = Vector3.new(camRight.X, 0, camRight.Z).Unit

            desiredVelocity = (right * input.X + forward * (-input.Z) + Vector3.new(0, input.Y, 0)) * flySpeed
        end

        bv.Velocity = bv.Velocity:Lerp(desiredVelocity, 0.25)

        local lookXZ = Vector3.new(camera.CFrame.LookVector.X, 0, camera.CFrame.LookVector.Z)
        if lookXZ.Magnitude > 0.001 then
            local targetCF = CFrame.lookAt(root.Position, root.Position + lookXZ.Unit)
            bg.CFrame = bg.CFrame:Lerp(targetCF, 0.2)
        end

        local speedMag = bv.Velocity.Magnitude
        local alpha = math.clamp(speedMag / (flySpeed * 1.2), 0, 1)
        local targetFOV = defaultFOV + (maxFOV - defaultFOV) * alpha
        camera.FieldOfView = camera.FieldOfView + (targetFOV - camera.FieldOfView) * fovSmooth
    end)
end

local function stopFly()
    if not flying then return end
    flying = false
    if flyConnection then
        flyConnection:Disconnect()
        flyConnection = nil
    end
    local char = player.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            for _, obj in ipairs(root:GetChildren()) do
                if obj:IsA("BodyGyro") or obj:IsA("BodyVelocity") then
                    obj:Destroy()
                end
            end
        end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.PlatformStand = false
            hum.AutoRotate = true
        end
    end
    if camera then
        camera.FieldOfView = 70
    end
end

        if state then
            startFly()
        else
            stopFly()
        end
    end
